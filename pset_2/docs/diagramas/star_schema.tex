% Diagrama del modelo dimensional (capa GOLD) del PSet 2. Se incluye con \input.
\begin{tikzpicture}[
    font=\scriptsize,
    every node/.append style={execute at begin node=\hyphenpenalty10000\relax},
    tabla/.style={draw=acento, rounded corners=2pt, fill=white, align=left, inner sep=2.5pt,
                  text width=#1},
    tabla/.default=24mm,
    hecho/.style={tabla=#1, fill=filaclara, very thick},
    hecho/.default=33mm,
    fk/.style={-{Stealth[length=1.8mm]}, acento},
  ]
  \newcommand{\tit}[1]{\textbf{\footnotesize\texttt{#1}}\\[-2pt]\rule{\linewidth}{0.3pt}\\[-1pt]}
  \newcommand{\pk}[1]{\texttt{#1}\,\textcolor{acento}{\tiny\textbf{PK}}}

  % ------------- hechos
  \node[hecho] (fa) at (2.2, 0) {\tit{fact\_awards}
     \textit{Grain: un award otorgado}\\
     \pk{award\_key}\\
     FK: funder, topic, institution, start\_date, end\_date\\[1pt]
     \textbf{Métricas:} amount\_usd, duration\_days, funded\_outputs\_count, award\_count};
  \node[hecho] (faw) at (9.1, 0) {\tit{fact\_award\_works}
     \textit{Grain: una publicación financiada (award × work)}\\
     \pk{award\_key + work\_key}\\
     FK: award, work, funder, topic, publication\_date (award también → fact\_awards)\\[1pt]
     \textbf{Sin métricas} (factless): funded\_work\_count = 1};

  % ------------- dimensiones compartidas (columna central)
  \node[tabla=25mm] (daw) at (5.65, 3.05) {\tit{dim\_award}\pk{award\_key}\\ título, descripción\\ (texto a vectorizar)};
  \node[tabla=25mm] (dfu) at (5.65, 1.0) {\tit{dim\_funder}\pk{funder\_key}\\ nombre, país, sur global};
  \node[tabla=25mm] (dto) at (5.65, -1.0) {\tit{dim\_topic}\pk{topic\_key}\\ topic › subfield › field › domain};
  \node[tabla=25mm] (dda) at (5.65, -3.05) {\tit{dim\_date}\pk{date\_key}\\ YYYYMMDD; año, mes, año fiscal};

  % ------------- solo award / solo work
  \node[tabla=24mm] (din) at (-1.4, 0) {\tit{dim\_institution}\pk{institution\_key}\\ nombre, país, tipo};
  \node[tabla=26mm] (dwo) at (12.6, 1.7) {\tit{dim\_work}\pk{work\_key}\\ título, abstract, año, tipo, citas; FK topic\_key};
  \node[tabla=26mm] (brg) at (12.6, -0.3) {\tit{brg\_work\_author}\pk{work\_key + author\_key}\\ posición, correspondencia};
  \node[tabla=26mm] (dau) at (12.6, -2.1) {\tit{dim\_author}\pk{author\_key}\\ nombre, ORCID};

  % ------------- relaciones (FK → PK)
  \foreach \d in {daw, dfu, dto, dda} {
    \draw[fk] (fa) -- (\d);
    \draw[fk] (faw) -- (\d);
  }
  \draw[fk] (fa) -- (din);
  \draw[fk] (faw) -- (dwo);
  \draw[fk] (brg) -- (dwo);
  \draw[fk] (brg) -- (dau);
\end{tikzpicture}
